class Alarm {
  final int id;
  final int hour;
  final int minute;
  final String label;
  final bool enabled;
  final List<int> repeatDays; // 1=Mon .. 7=Sun; empty = Once
  final bool vibrate;
  final int snoozeMinutes;
  final bool fadeIn;
  final String? soundPath; // null = default alarm sound

  const Alarm({
    required this.id,
    required this.hour,
    required this.minute,
    this.label = '',
    this.enabled = true,
    this.repeatDays = const [],
    this.vibrate = true,
    this.snoozeMinutes = 10,
    this.fadeIn = true,
    this.soundPath,
  });

  Alarm copyWith({
    int? id,
    int? hour,
    int? minute,
    String? label,
    bool? enabled,
    List<int>? repeatDays,
    bool? vibrate,
    int? snoozeMinutes,
    bool? fadeIn,
    String? soundPath,
    bool clearSoundPath = false,
  }) {
    return Alarm(
      id: id ?? this.id,
      hour: hour ?? this.hour,
      minute: minute ?? this.minute,
      label: label ?? this.label,
      enabled: enabled ?? this.enabled,
      repeatDays: repeatDays ?? this.repeatDays,
      vibrate: vibrate ?? this.vibrate,
      snoozeMinutes: snoozeMinutes ?? this.snoozeMinutes,
      fadeIn: fadeIn ?? this.fadeIn,
      soundPath: clearSoundPath ? null : (soundPath ?? this.soundPath),
    );
  }

  DateTime nextTriggerTime([DateTime? from]) {
    final now = from ?? DateTime.now();
    var candidate = DateTime(now.year, now.month, now.day, hour, minute);

    if (repeatDays.isEmpty) {
      if (!candidate.isAfter(now)) {
        candidate = candidate.add(const Duration(days: 1));
      }
      return candidate;
    }

    for (int offset = 0; offset <= 7; offset++) {
      final day = candidate.add(Duration(days: offset));
      if (repeatDays.contains(day.weekday) && day.isAfter(now)) {
        return day;
      }
    }
    return candidate.add(const Duration(days: 1));
  }

  String formattedTime(BuildContext context) {
    return TimeOfDay(hour: hour, minute: minute).format(context);
  }

  String get repeatSummary {
    if (repeatDays.isEmpty) return 'Once';
    if (repeatDays.length == 7) return 'Daily';
    const names = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    final sorted = [...repeatDays]..sort();
    return sorted.map((d) => names[d - 1]).join(', ');
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'hour': hour,
        'minute': minute,
        'label': label,
        'enabled': enabled,
        'repeatDays': repeatDays,
        'vibrate': vibrate,
        'snoozeMinutes': snoozeMinutes,
        'fadeIn': fadeIn,
        'soundPath': soundPath,
      };

  factory Alarm.fromJson(Map<String, dynamic> json) => Alarm(
        id: json['id'] as int,
        hour: json['hour'] as int,
        minute: json['minute'] as int,
        label: json['label'] as String? ?? '',
        enabled: json['enabled'] as bool? ?? true,
        repeatDays: (json['repeatDays'] as List<dynamic>? ?? [])
            .map((e) => e as int)
            .toList(),
        vibrate: json['vibrate'] as bool? ?? true,
        snoozeMinutes: json['snoozeMinutes'] as int? ?? 10,
        fadeIn: json['fadeIn'] as bool? ?? true,
        soundPath: json['soundPath'] as String?,
      );
}

late final AppAudioHandler audioHandler;

Future<void> initAudioHandler() async {
  audioHandler = await AudioService.init(
    builder: () => AppAudioHandler(),
    config: const AudioServiceConfig(
      androidNotificationChannelId: 'com.example.music_alarm_app.audio',
      androidNotificationChannelName: 'Music Playback',
      androidNotificationOngoing: true,
      androidStopForegroundOnPause: true,
    ),
  );
}

class AppAudioHandler extends BaseAudioHandler with QueueHandler, SeekHandler {
  final AudioPlayer player = AudioPlayer();

  final ValueNotifier<List<Track>> currentTracks = ValueNotifier([]);
  final ValueNotifier<Duration?> sleepRemaining = ValueNotifier(null);
  final ValueNotifier<bool> stopAfterCurrent = ValueNotifier(false);
  final ValueNotifier<bool> shuffleEnabled = ValueNotifier(false);
  final ValueNotifier<LoopMode> loopMode = ValueNotifier(LoopMode.off);

  Timer? _sleepTimer;
  bool _askedNotificationOnPlay = false;
  int? _lastIndex;

  AppAudioHandler() {
    player.playbackEventStream.map(_transformEvent).pipe(playbackState);

    player.currentIndexStream.listen((index) async {
      final list = currentTracks.value;
      if (index != null && index >= 0 && index < list.length) {
        if (stopAfterCurrent.value && _lastIndex != null && _lastIndex != index) {
          stopAfterCurrent.value = false;
          await pause();
        }
        _lastIndex = index;
        mediaItem.add(list[index].toMediaItem());
      }
    });

    player.processingStateStream.listen((state) {
      if (state == ProcessingState.completed && stopAfterCurrent.value) {
        stopAfterCurrent.value = false;
      }
    });
  }

  PlaybackState _transformEvent(PlaybackEvent event) {
    return PlaybackState(
      controls: [
        MediaControl.skipToPrevious,
        if (player.playing) MediaControl.pause else MediaControl.play,
        MediaControl.skipToNext,
      ],
      systemActions: const {
        MediaAction.seek,
        MediaAction.seekForward,
        MediaAction.seekBackward,
      },
      androidCompactActionIndices: const [0, 1, 2],
      processingState: const {
        ProcessingState.idle: AudioProcessingState.idle,
        ProcessingState.loading: AudioProcessingState.loading,
        ProcessingState.buffering: AudioProcessingState.buffering,
        ProcessingState.ready: AudioProcessingState.ready,
        ProcessingState.completed: AudioProcessingState.completed,
      }[player.processingState]!,
      playing: player.playing,
      updatePosition: player.position,
      bufferedPosition: player.bufferedPosition,
      speed: player.speed,
      queueIndex: event.currentIndex,
      shuffleMode: shuffleEnabled.value
          ? AudioServiceShuffleMode.all
          : AudioServiceShuffleMode.none,
      repeatMode: const {
        LoopMode.off: AudioServiceRepeatMode.none,
        LoopMode.all: AudioServiceRepeatMode.all,
        LoopMode.one: AudioServiceRepeatMode.one,
      }[loopMode.value]!,
    );
  }

  Future<void> _ensureNotificationPermission() async {
    if (_askedNotificationOnPlay) return;
    _askedNotificationOnPlay = true;
    if (await Permission.notification.isDenied) {
      await Permission.notification.request();
    }
  }

  Future<void> setQueueAndPlay(List<Track> tracks, int initialIndex) async {
    if (tracks.isEmpty) return;
    await _ensureNotificationPermission();

    _lastIndex = null;
    currentTracks.value = List<Track>.from(tracks);
    queue.add(tracks.map((t) => t.toMediaItem()).toList());

    final sources = tracks.map((t) => AudioSource.file(t.path)).toList();
    await player.setAudioSources(
      sources,
      initialIndex: initialIndex.clamp(0, tracks.length - 1),
      initialPosition: Duration.zero,
    );
    mediaItem.add(tracks[initialIndex.clamp(0, tracks.length - 1)].toMediaItem());
    await play();
  }

  Future<void> playNext(Track track) async {
    final list = [...currentTracks.value];
    if (list.isEmpty) {
      await setQueueAndPlay([track], 0);
      return;
    }
    final insertIdx = (player.currentIndex ?? 0) + 1;
    list.insert(insertIdx, track);
    currentTracks.value = list;
    queue.add(list.map((t) => t.toMediaItem()).toList());
    await player.insertAudioSource(insertIdx, AudioSource.file(track.path));
  }

  Future<void> addToQueue(Track track) async {
    final list = [...currentTracks.value];
    if (list.isEmpty) {
      await setQueueAndPlay([track], 0);
      return;
    }
    list.add(track);
    currentTracks.value = list;
    queue.add(list.map((t) => t.toMediaItem()).toList());
    await player.addAudioSource(AudioSource.file(track.path));
  }

  Future<void> reorderQueue(int oldIndex, int newIndex) async {
    final list = [...currentTracks.value];
    if (oldIndex < 0 || oldIndex >= list.length) return;
    if (oldIndex < newIndex) newIndex -= 1;
    if (newIndex < 0 || newIndex >= list.length) return;

    final item = list.removeAt(oldIndex);
    list.insert(newIndex, item);
    currentTracks.value = list;
    queue.add(list.map((t) => t.toMediaItem()).toList());
    await player.moveAudioSource(oldIndex, newIndex);
  }

  Future<void> toggleShuffle() async {
    final next = !shuffleEnabled.value;
    shuffleEnabled.value = next;
    await player.setShuffleModeEnabled(next);
    if (next) await player.shuffle();
  }

  Future<void> cycleRepeatMode() async {
    final next = switch (loopMode.value) {
      LoopMode.off => LoopMode.all,
      LoopMode.all => LoopMode.one,
      LoopMode.one => LoopMode.off,
    };
    loopMode.value = next;
    await player.setLoopMode(next);
  }

  // Sleep timer controls
  void startSleepTimer(Duration duration) {
    _sleepTimer?.cancel();
    sleepRemaining.value = duration;
    _sleepTimer = Timer.periodic(const Duration(seconds: 1), (timer) async {
      final rem = sleepRemaining.value;
      if (rem == null) {
        timer.cancel();
        return;
      }
      final next = rem - const Duration(seconds: 1);
      if (next <= Duration.zero) {
        timer.cancel();
        sleepRemaining.value = null;
        stopAfterCurrent.value = false;
        await pause();
      } else {
        sleepRemaining.value = next;
      }
    });
  }

  void extendSleepTimer(Duration extra) {
    final current = sleepRemaining.value;
    if (current != null) {
      sleepRemaining.value = current + extra;
    } else {
      startSleepTimer(extra);
    }
  }

  void cancelSleepTimer() {
    _sleepTimer?.cancel();
    _sleepTimer = null;
    sleepRemaining.value = null;
    stopAfterCurrent.value = false;
  }

  void toggleStopAfterCurrent() {
    stopAfterCurrent.value = !stopAfterCurrent.value;
  }

  @override
  Future<void> play() async {
    await _ensureNotificationPermission();
    await player.play();
  }

  @override
  Future<void> pause() => player.pause();

  @override
  Future<void> stop() async {
    cancelSleepTimer();
    await player.stop();
    await super.stop();
  }

  @override
  Future<void> seek(Duration position) => player.seek(position);

  @override
  Future<void> skipToNext() => player.seekToNext();

  @override
  Future<void> skipToPrevious() async {
    if (player.position > const Duration(seconds: 3)) {
      await player.seek(Duration.zero);
    } else {
      await player.seekToPrevious();
    }
  }

  @override
  Future<void> skipToQueueItem(int index) async {
    await player.seek(Duration.zero, index: index);
    await play();
  }
}

const String _alarmChannelId = 'alarm_ring_channel';
const String _alarmPortName = 'music_alarm_ring_port';

@pragma('vm:entry-point')

@pragma('vm:entry-point')
Future<void> alarmCallback(int id, Map<String, dynamic> params) async {
  DartPluginRegistrant.ensureInitialized();
  WidgetsFlutterBinding.ensureInitialized();

  final plugin = FlutterLocalNotificationsPlugin();
  const androidInit = AndroidInitializationSettings('@mipmap/ic_launcher');
  await plugin.initialize(
    settings: const InitializationSettings(android: androidInit),
  );

  final alarm = Alarm.fromJson(params);
  final title = alarm.label.isEmpty ? 'Alarm' : alarm.label;
  final body =
      '${alarm.hour.toString().padLeft(2, '0')}:${alarm.minute.toString().padLeft(2, '0')}';

  final details = AndroidNotificationDetails(
    _alarmChannelId,
    'Alarms',
    channelDescription: 'Full-screen alarm alerts',
    importance: Importance.max,
    priority: Priority.max,
    fullScreenIntent: true,
    category: AndroidNotificationCategory.alarm,
    audioAttributesUsage: AudioAttributesUsage.alarm,
    ongoing: true,
    autoCancel: false,
    enableVibration: alarm.vibrate,
    actions: const [
      AndroidNotificationAction('stop', 'Stop', showsUserInterface: true),
      AndroidNotificationAction('snooze', 'Snooze', showsUserInterface: true),
    ],
  );

  await plugin.show(
    id: id,
    title: title,
    body: body,
    notificationDetails: NotificationDetails(android: details),
    payload: jsonEncode(params),
  );

  // If the main UI isolate is alive, notify it immediately.
  final SendPort? sendPort = IsolateNameServer.lookupPortByName(_alarmPortName);
  sendPort?.send(jsonEncode(params));
}

class AlarmService {
  final ValueNotifier<List<Alarm>> alarms = ValueNotifier([]);
  final ValueNotifier<Alarm?> ringingAlarm = ValueNotifier(null);

  final FlutterLocalNotificationsPlugin _notifications =
      FlutterLocalNotificationsPlugin();
  final AudioPlayer _ringPlayer = AudioPlayer();

  GlobalKey<NavigatorState>? _navKey;
  ReceivePort? _receivePort;
  Timer? _fadeTimer;
  Timer? _vibrateTimer;

  Future<void> init(GlobalKey<NavigatorState> navKey) async {
    _navKey = navKey;
    await AndroidAlarmManager.initialize();

    _receivePort?.close();
    _receivePort = ReceivePort();
    IsolateNameServer.removePortNameMapping(_alarmPortName);
    IsolateNameServer.registerPortWithName(_receivePort!.sendPort, _alarmPortName);
    _receivePort!.listen((message) {
      if (message is String) {
        final alarm = Alarm.fromJson(jsonDecode(message) as Map<String, dynamic>);
        startRinging(alarm);
      }
    });

    const androidInit = AndroidInitializationSettings('@mipmap/ic_launcher');
    await _notifications.initialize(
      settings: const InitializationSettings(android: androidInit),
      onDidReceiveNotificationResponse: _onNotificationResponse,
      onDidReceiveBackgroundNotificationResponse: notificationTapBackground,
    );

    const channel = AndroidNotificationChannel(
      _alarmChannelId,
      'Alarms',
      description: 'Full-screen alarm alerts',
      importance: Importance.max,
      audioAttributesUsage: AudioAttributesUsage.alarm,
    );
    await _notifications
        .resolvePlatformSpecificImplementation<
            AndroidFlutterLocalNotificationsPlugin>()
        ?.createNotificationChannel(channel);

    await loadAlarms();
    await _rearmEnabledAlarms();

    final launchDetails =
        await _notifications.getNotificationAppLaunchDetails();
    if (launchDetails?.didNotificationLaunchApp ?? false) {
      final response = launchDetails!.notificationResponse;
      if (response != null) {
        _onNotificationResponse(response);
      }
    }
  }

  void _onNotificationResponse(NotificationResponse response) {
    final payload = response.payload;
    if (payload == null || payload.isEmpty) return;
    final alarm = Alarm.fromJson(jsonDecode(payload) as Map<String, dynamic>);

    if (response.actionId == 'stop') {
      ringingAlarm.value = alarm;
      stopRinging();
    } else if (response.actionId == 'snooze') {
      ringingAlarm.value = alarm;
      snoozeRinging();
    } else {
      startRinging(alarm);
    }
  }

  Future<void> loadAlarms() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_alarmsPrefsKey);
    if (raw == null) {
      alarms.value = [];
      return;
    }
    final list = (jsonDecode(raw) as List<dynamic>)
        .map((e) => Alarm.fromJson(e as Map<String, dynamic>))
        .toList();
    list.sort((a, b) => (a.hour * 60 + a.minute).compareTo(b.hour * 60 + b.minute));
    alarms.value = list;
  }

  Future<void> saveAlarms() async {
    final prefs = await SharedPreferences.getInstance();
    final encoded = jsonEncode(alarms.value.map((a) => a.toJson()).toList());
    await prefs.setString(_alarmsPrefsKey, encoded);
  }

  Future<void> _rearmEnabledAlarms() async {
    for (final alarm in alarms.value.where((a) => a.enabled)) {
      await _scheduleOneShot(alarm);
    }
  }

  Future<bool> _requestAlarmPermissions() async {
    final exactStatus = await Permission.scheduleExactAlarm.status;
    if (!exactStatus.isGranted) {
      await Permission.scheduleExactAlarm.request();
    }
    if (await Permission.notification.isDenied) {
      await Permission.notification.request();
    }
    await _notifications
        .resolvePlatformSpecificImplementation<
            AndroidFlutterLocalNotificationsPlugin>()
        ?.requestFullScreenIntentPermission();
    return await Permission.scheduleExactAlarm.isGranted;
  }

  Future<void> scheduleAlarm(Alarm alarm) async {
    await _requestAlarmPermissions();
    await _scheduleOneShot(alarm);
  }

  Future<void> _scheduleOneShot(Alarm alarm) async {
    final triggerAt = alarm.nextTriggerTime();
    await AndroidAlarmManager.oneShotAt(
      triggerAt,
      alarm.id,
      alarmCallback,
      exact: true,
      wakeup: true,
      allowWhileIdle: true,
      alarmClock: true,
      rescheduleOnReboot: true,
      params: alarm.toJson(),
    );
  }

  Future<void> cancelAlarm(int id) async {
    await AndroidAlarmManager.cancel(id);
    await AndroidAlarmManager.cancel(id + 1000000); // snooze id
    await _notifications.cancel(id: id);
  }

  Future<void> upsertAlarm(Alarm alarm) async {
    final list = [...alarms.value];
    final idx = list.indexWhere((a) => a.id == alarm.id);
    if (idx >= 0) {
      list[idx] = alarm;
    } else {
      list.add(alarm);
    }
    list.sort((a, b) => (a.hour * 60 + a.minute).compareTo(b.hour * 60 + b.minute));
    alarms.value = list;
    await saveAlarms();

    if (alarm.enabled) {
      await scheduleAlarm(alarm);
    } else {
      await cancelAlarm(alarm.id);
    }
  }

  Future<void> toggleAlarm(Alarm alarm, bool enabled) async {
    await upsertAlarm(alarm.copyWith(enabled: enabled));
  }

  Future<void> deleteAlarm(int id) async {
    await cancelAlarm(id);
    alarms.value = alarms.value.where((a) => a.id != id).toList();
    await saveAlarms();
  }

  (Alarm, DateTime)? nextEnabledAlarm() {
    final enabled = alarms.value.where((a) => a.enabled).toList();
    if (enabled.isEmpty) return null;
    final now = DateTime.now();
    Alarm? best;
    DateTime? bestTime;
    for (final a in enabled) {
      final t = a.nextTriggerTime(now);
      if (bestTime == null || t.isBefore(bestTime)) {
        best = a;
        bestTime = t;
      }
    }
    if (best == null || bestTime == null) return null;
    return (best, bestTime);
  }

  Future<void> startRinging(Alarm alarm) async {
    ringingAlarm.value = alarm;
    _navKey?.currentState?.pushNamed('/ring');

    _fadeTimer?.cancel();
    _vibrateTimer?.cancel();

    try {
      await _ringPlayer.stop();
      if (alarm.soundPath != null) {
        await _ringPlayer.setAudioSource(AudioSource.file(alarm.soundPath!));
      } else {
        await _ringPlayer.setAudioSource(
          AudioSource.uri(Uri.parse('content://settings/system/alarm_alert')),
        );
      }
      await _ringPlayer.setLoopMode(LoopMode.one);
      if (alarm.fadeIn) {
        double vol = 0.05;
        await _ringPlayer.setVolume(vol);
        _fadeTimer = Timer.periodic(const Duration(seconds: 1), (t) async {
          vol = (vol + (0.95 / 30)).clamp(0.05, 1.0);
          await _ringPlayer.setVolume(vol);
          if (vol >= 1.0) t.cancel();
        });
      } else {
        await _ringPlayer.setVolume(1.0);
      }
      await _ringPlayer.play();
    } catch (_) {
      // Fallback: notification channel alarm sound still rings.
    }

    if (alarm.vibrate) {
      _vibrateTimer = Timer.periodic(const Duration(seconds: 1), (_) {
        HapticFeedback.vibrate();
      });
    }
  }

  Future<void> stopRinging() async {
    final alarm = ringingAlarm.value;
    _fadeTimer?.cancel();
    _vibrateTimer?.cancel();
    await _ringPlayer.stop();
    if (alarm != null) {
      await _notifications.cancel(id: alarm.id);
      if (alarm.repeatDays.isEmpty) {
        await upsertAlarm(alarm.copyWith(enabled: false));
      } else {
        await _scheduleOneShot(alarm);
      }
    }
    ringingAlarm.value = null;
    if (_navKey?.currentState?.canPop() ?? false) {
      _navKey?.currentState?.pop();
    }
  }

  Future<void> snoozeRinging() async {
    final alarm = ringingAlarm.value;
    _fadeTimer?.cancel();
    _vibrateTimer?.cancel();
    await _ringPlayer.stop();
    if (alarm != null) {
      await _notifications.cancel(id: alarm.id);
      final snoozeTime =
          DateTime.now().add(Duration(minutes: alarm.snoozeMinutes));
      await AndroidAlarmManager.oneShotAt(
        snoozeTime,
        alarm.id + 1000000,
        alarmCallback,
        exact: true,
        wakeup: true,
        allowWhileIdle: true,
        alarmClock: true,
        params: alarm.toJson(),
      );
    }
    ringingAlarm.value = null;
    if (_navKey?.currentState?.canPop() ?? false) {
      _navKey?.currentState?.pop();
    }
  }
}

  Future<void> _scanFolders() async {
    setState(() => _scanning = true);
    final List<Track> found = [];
    final Set<String> seenPaths = {};

    for (final folder in _folders) {
      final dir = Directory(folder);
      if (!await dir.exists()) continue;
      try {
        await for (final entity in dir.list(recursive: true, followLinks: false)) {
          if (entity is! File) continue;
          final lower = entity.path.toLowerCase();
          final ok = _supportedExts.any((ext) => lower.endsWith(ext));
          if (!ok || seenPaths.contains(entity.path)) continue;
          seenPaths.add(entity.path);

          final fileName = entity.uri.pathSegments.last;
          final fallbackTitle = fileName.contains('.')
              ? fileName.substring(0, fileName.lastIndexOf('.'))
              : fileName;

          String title = fallbackTitle;
          String artist = 'Unknown Artist';
          String album = 'Unknown Album';
          int durationMs = 0;
          int modifiedMs = 0;

          try {
            final stat = await entity.stat();
            modifiedMs = stat.modified.millisecondsSinceEpoch;
            final meta = readMetadata(entity, getImage: false);
            if ((meta.title ?? '').trim().isNotEmpty) {
              title = meta.title!.trim();
            }
            if ((meta.artist ?? '').trim().isNotEmpty) {
              artist = meta.artist!.trim();
            }
            if ((meta.album ?? '').trim().isNotEmpty) {
              album = meta.album!.trim();
            }
            if (meta.duration != null) {
              durationMs = meta.duration!.inMilliseconds;
            }
          } catch (_) {
            // Unreadable tags: keep filename fallback.
          }

          found.add(Track(
            id: entity.path,
            path: entity.path,
            title: title,
            artist: artist,
            album: album,
            durationMs: durationMs,
            dateAddedMs: modifiedMs,
          ));
        }
      } catch (_) {
        // Skip unreadable directory.
      }
    }

    libraryTracks.value = found;
    if (mounted) {
      setState(() => _scanning = false);
    }

                    final maxSec = total.inSeconds.toDouble().clamp(1.0, double.infinity);
                    final currentSec = (_dragSeconds ?? pos.inSeconds.toDouble())
                        .clamp(0.0, maxSec);

                    return Column(
                      children: [
                        Slider(
                          value: currentSec,
                          max: maxSec,
                          onChanged: (v) => setState(() => _dragSeconds = v),
                          onChangeEnd: (v) async {
                            await audioHandler.seek(Duration(seconds: v.round()));
                            if (mounted) setState(() => _dragSeconds = null);
                          },
                        ),
                        Row(
                          mainAxisAlignment: MainAxisAlignment.spaceBetween,
                          children: [
                            Text(_fmt(Duration(seconds: currentSec.round()))),
                            Text(_fmt(total)),
                          ],
                        ),
                      ],
                    );
                  },
                ),
                const SizedBox(height: 12),
                StreamBuilder<PlaybackState>(
                  stream: audioHandler.playbackState,
                  builder: (context, stateSnap) {
                    final playing = stateSnap.data?.playing ?? false;
                    return Row(
                      mainAxisAlignment: MainAxisAlignment.spaceEvenly,
                      children: [
                        ValueListenableBuilder<bool>(
                          valueListenable: audioHandler.shuffleEnabled,
                          builder: (_, shuf, __) => IconButton(
                            icon: Icon(
                              Icons.shuffle,
                              color: shuf
                                  ? Theme.of(context).colorScheme.primary
                                  : null,
                            ),
                            onPressed: audioHandler.toggleShuffle,
                          ),
                        ),
                        IconButton(
                          iconSize: 36,
                          icon: const Icon(Icons.skip_previous),
                          onPressed: audioHandler.skipToPrevious,
                        ),
                        FilledButton(
                          style: FilledButton.styleFrom(
                            shape: const CircleBorder(),
                            padding: const EdgeInsets.all(18),
                          ),
                          onPressed:
                              playing ? audioHandler.pause : audioHandler.play,
                          child: Icon(
                            playing ? Icons.pause : Icons.play_arrow,
                            size: 36,
                          ),
                        ),
                        IconButton(
                          iconSize: 36,
                          icon: const Icon(Icons.skip_next),
                          onPressed: audioHandler.skipToNext,
                        ),
                        ValueListenableBuilder<LoopMode>(
                          valueListenable: audioHandler.loopMode,
                          builder: (_, mode, __) => IconButton(
                            icon: Icon(
                              mode == LoopMode.one
                                  ? Icons.repeat_one
                                  : Icons.repeat,
                              color: mode != LoopMode.off
                                  ? Theme.of(context).colorScheme.primary
                                  : null,
                            ),
                            onPressed: audioHandler.cycleRepeatMode,
                          ),
                        ),
                      ],
                    );
                  },
                ),
                const SizedBox(height: 12),
              ],
            ),
          );
        },
      ),
    );
